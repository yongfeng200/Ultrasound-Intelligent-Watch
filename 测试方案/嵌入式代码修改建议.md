1. **输出结构化测量数据** - 便于 MATLAB 回放和分析
2. **添加调试模式** - 方便开发和调试
3. **支持 Golden Vector 验证** - 确保算法实现正确性
4. **提供统计信息** - 实时监控系统状态

```c
#ifndef __MEASUREMENT_H__
#define __MEASUREMENT_H__

#include <stdint.h>

/* 单次测量数据 */
typedef struct {
    uint32_t timestamp_ms;      // 时间戳 (ms)
    uint32_t tof_down_ticks;    // 顺流 TOF (ticks)
    uint32_t tof_up_ticks;      // 逆流 TOF (ticks)
    float    temperature_c;     // 温度 (°C)
    uint8_t  valid_down;        // 顺流有效标志
    uint8_t  valid_up;          // 逆流有效标志
    uint8_t  status;            // 状态字节
} measurement_data_t;

/* 计算结果 */
typedef struct {
    float velocity_m_s;         // 流速 (m/s)
    float flow_l_min;           // 流量 (L/min)
    float volume_l;             // 累计体积 (L)
    uint8_t valid;              // 结果有效标志
} flow_result_t;

/* 统计信息 */
typedef struct {
    uint32_t total_count;       // 总测量次数
    uint32_t valid_count;       // 有效测量次数
    uint32_t timeout_count;     // 超时次数
    uint32_t outlier_count;     // 异常值次数
    float    avg_velocity;      // 平均流速
    float    avg_flow;          // 平均流量
} measurement_stats_t;

/* 函数声明 */
void measurement_init(void);
int  measurement_acquire(measurement_data_t *data);
int  measurement_calculate(const measurement_data_t *data, flow_result_t *result);
void measurement_update_stats(const measurement_data_t *data, const flow_result_t *result);
void measurement_print_stats(void);

#endif /* __MEASUREMENT_H__ */
```

```c
#include "config.h"
#include "measurement.h"

/**
 * @brief 输出测量数据（CSV 格式）
 * @param data 测量数据指针
 * @param result 计算结果指针
 */
void uart_output_measurement_csv(const measurement_data_t *data, 
                                 const flow_result_t *result)
{
#if CSV_OUTPUT_MODE
    // CSV 格式：便于 Python 和 MATLAB 解析
    // 格式：MEASURE,timestamp,tof_down,tof_up,temp,valid_down,valid_up,velocity,flow,volume
    printf("MEASURE,%u,%u,%u,%.2f,%u,%u,%.6f,%.3f,%.3f\n",
           data->timestamp_ms,
           data->tof_down_ticks,
           data->tof_up_ticks,
           data->temperature_c,
           data->valid_down,
           data->valid_up,
           result->velocity_m_s,
           result->flow_l_min,
           result->volume_l);
#else
    // 人类可读格式
    printf("[%6u ms] TOF: %u / %u ticks, Temp: %.2f°C, "
           "Velocity: %.3f m/s, Flow: %.2f L/min, Volume: %.2f L\n",
           data->timestamp_ms,
           data->tof_down_ticks,
           data->tof_up_ticks,
           data->temperature_c,
           result->velocity_m_s,
           result->flow_l_min,
           result->volume_l);
#endif
}

/**
 * @brief 输出统计信息
 * @param stats 统计信息指针
 */
void uart_output_statistics(const measurement_stats_t *stats)
{
#if PRINT_STATISTICS
    float valid_rate = (stats->total_count > 0) ? 
                       (100.0f * stats->valid_count / stats->total_count) : 0.0f;
    
    printf("STATS: Total=%u, Valid=%u, Rate=%.2f%%, "
           "Timeout=%u, Outlier=%u, AvgFlow=%.2f L/min\n",
           stats->total_count,
           stats->valid_count,
           valid_rate,
           stats->timeout_count,
           stats->outlier_count,
           stats->avg_flow);
#endif
}

/**
 * @brief 输出表头（CSV 模式）
 */
void uart_output_csv_header(void)
{
#if CSV_OUTPUT_MODE
    printf("# LS1D Ultrasonic Water Meter Data\n");
    printf("# Format: MEASURE,timestamp_ms,tof_down_ticks,tof_up_ticks,"
           "temperature_c,valid_down,valid_up,velocity_m_s,flow_l_min,volume_l\n");
    printf("# TDC tick period: %.1f ns\n", TDC_TICK_PERIOD_NS);
    printf("# Pipe diameter: %.1f mm\n", PIPE_DIAMETER_MM);
    printf("# Sound path: %.1f mm\n", SOUND_PATH_MM);
    printf("# Install angle: %.1f deg\n", INSTALL_ANGLE_DEG);
    printf("START\n");
#endif
}
```

### 4. 修改主循环

**修改文件：** `UserSrc/main.c`

```c
#include "config.h"
#include "measurement.h"

/* 全局变量 */
static measurement_stats_t g_stats = {0};
static uint32_t g_measurement_count = 0;

/**
 * @brief 测量任务主循环
 */
void measurement_task(void)
{
    measurement_data_t data;
    flow_result_t result;
    
    // 初始化
    measurement_init();
    uart_output_csv_header();
    
    printf("Measurement started. Period: %u ms\n", MEASUREMENT_PERIOD_MS);
    
    while (1) {
        // 采集测量数据
        if (measurement_acquire(&data) == 0) {
            
            // 计算流速和流量
            measurement_calculate(&data, &result);
            
            // 输出数据
            uart_output_measurement_csv(&data, &result);
            
            // 更新统计
            measurement_update_stats(&data, &result);
            g_measurement_count++;
            
            // 定期输出统计信息
            if (g_measurement_count % STATS_INTERVAL_COUNT == 0) {
                uart_output_statistics(&g_stats);
            }
        }
        
        // 延时到下一个测量周期
        delay_ms(MEASUREMENT_PERIOD_MS);
    }
}

int main(void)
{
    // 系统初始化
    system_init();
    uart_init(115200);
    
#if GOLDEN_VECTOR_TEST
    // Golden Vector 测试模式
    run_golden_vector_test();
#else
    // 正常测量模式
    measurement_task();
#endif
    
    return 0;
}
```

### 5. 实现测量和计算函数

**新建文件：** `UserSrc/measurement.c`

```c
#include "measurement.h"
#include "config.h"
#include "ls1d_std_measure.h"
#include <math.h>

/* 常量定义 */
#define PI 3.14159265359f
#define SOUND_SPEED_20C  1482.0f      // 20°C 时声速 (m/s)
#define SOUND_SPEED_SLOPE 2.5f        // 声速温度系数 (m/s/°C)

/* 全局变量 */
static float g_cumulative_volume_l = 0.0f;
static uint32_t g_last_timestamp_ms = 0;

/* 滤波缓冲区 */
#if ENABLE_FILTER
static float g_velocity_buffer[FILTER_WINDOW_SIZE] = {0};
static uint8_t g_buffer_index = 0;
static uint8_t g_buffer_filled = 0;
#endif

/**
 * @brief 初始化测量模块
 */
void measurement_init(void)
{
    // 初始化测量硬件
    MEASURE_DefaultInit();
    MEASURE_GateWindowInit();
    
    g_cumulative_volume_l = 0.0f;
    g_last_timestamp_ms = 0;
    
#if ENABLE_FILTER
    for (int i = 0; i < FILTER_WINDOW_SIZE; i++) {
        g_velocity_buffer[i] = 0.0f;
    }
    g_buffer_index = 0;
    g_buffer_filled = 0;
#endif
}

/**
 * @brief 采集一次测量数据
 * @param data 输出：测量数据
 * @return 0: 成功, -1: 失败
 */
int measurement_acquire(measurement_data_t *data)
{
    // 触发测量
    // 这里需要根据实际的测量驱动接口调用
    // 示例：
    uint32_t tdc_raw_down, tdc_raw_up;
    uint8_t valid_flags;
    
    // 调用底层测量函数（需要根据实际 API 调整）
    if (sonar_measure(&tdc_raw_down, &tdc_raw_up, &valid_flags) != 0) {
        return -1;
    }
    
    // 填充数据结构
    data->timestamp_ms = get_system_time_ms();  // 需要实现系统时钟
    data->tof_down_ticks = tdc_raw_down;
    data->tof_up_ticks = tdc_raw_up;
    data->temperature_c = read_temperature();    // 需要实现温度读取
    data->valid_down = (valid_flags & 0x01) ? 1 : 0;
    data->valid_up = (valid_flags & 0x02) ? 1 : 0;
    data->status = valid_flags;
    
    return 0;
}

/**
 * @brief 计算流速和流量
 * @param data 输入：测量数据
 * @param result 输出：计算结果
 * @return 0: 成功, -1: 无效数据
 */
int measurement_calculate(const measurement_data_t *data, flow_result_t *result)
{
    // 检查有效性
    if (!data->valid_down || !data->valid_up) {
        result->valid = 0;
        result->velocity_m_s = 0.0f;
        result->flow_l_min = 0.0f;
        result->volume_l = g_cumulative_volume_l;
        return -1;
    }
    
    // 转换为时间（秒）
    float tof_down_s = data->tof_down_ticks * TDC_TICK_PERIOD_NS * 1e-9f;
    float tof_up_s = data->tof_up_ticks * TDC_TICK_PERIOD_NS * 1e-9f;
    
    // 温度补偿
    float sound_speed = SOUND_SPEED_20C + 
                        SOUND_SPEED_SLOPE * (data->temperature_c - 20.0f);
    
    // 计算流速
    float sound_path_m = SOUND_PATH_MM / 1000.0f;
    float cos_theta = cosf(INSTALL_ANGLE_DEG * PI / 180.0f);
    
    // v = L / (2 * cos(θ)) * (1/t_down - 1/t_up)
    float velocity_raw = sound_path_m / (2.0f * cos_theta) * 
                         (1.0f / tof_down_s - 1.0f / tof_up_s);
    
#if ENABLE_FILTER
    // 中值滤波
    g_velocity_buffer[g_buffer_index] = velocity_raw;
    g_buffer_index = (g_buffer_index + 1) % FILTER_WINDOW_SIZE;
    if (g_buffer_index == 0) g_buffer_filled = 1;
    
    float velocity_filtered = median_filter(g_velocity_buffer, 
                                           g_buffer_filled ? FILTER_WINDOW_SIZE : g_buffer_index);
    result->velocity_m_s = velocity_filtered;
#else
    result->velocity_m_s = velocity_raw;
#endif
    
    // 计算流量 Q = v * A
    float pipe_radius_m = (PIPE_DIAMETER_MM / 1000.0f) / 2.0f;
    float area_m2 = PI * pipe_radius_m * pipe_radius_m;
    float flow_m3_s = result->velocity_m_s * area_m2;
    result->flow_l_min = flow_m3_s * 60000.0f;  // 转换为 L/min
    
    // 累计体积
    if (g_last_timestamp_ms > 0) {
        float delta_t_s = (data->timestamp_ms - g_last_timestamp_ms) / 1000.0f;
        float delta_volume_l = flow_m3_s * delta_t_s * 1000.0f;
        g_cumulative_volume_l += delta_volume_l;
    }
    g_last_timestamp_ms = data->timestamp_ms;
    result->volume_l = g_cumulative_volume_l;
    
    result->valid = 1;
    return 0;
}

/**
 * @brief 中值滤波
 */
static float median_filter(float *buffer, uint8_t length)
{
    // 简单冒泡排序
    float sorted[FILTER_WINDOW_SIZE];
    for (int i = 0; i < length; i++) {
        sorted[i] = buffer[i];
    }
    
    for (int i = 0; i < length - 1; i++) {
        for (int j = 0; j < length - i - 1; j++) {
            if (sorted[j] > sorted[j + 1]) {
                float temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }
    
    return sorted[length / 2];
}
```

### 6. Golden Vector 测试

**修改文件：** `UserSrc/golden_vector_test.c` (新建)

```c
#include "measurement.h"
#include "config.h"

#if GOLDEN_VECTOR_TEST

/* Golden Vector 测试数据 */
/* 这些数据从 MATLAB 的 golden_vectors/matlab_golden.csv 中获取 */
const uint32_t test_tof_down[] = {67500, 67505, 67502, 67498, 67503};
const uint32_t test_tof_up[] = {68200, 68195, 68198, 68202, 68197};
const float test_temp[] = {20.0, 20.1, 20.0, 20.0, 20.1};
const float expected_velocity[] = {0.123456, 0.125678, 0.124012, 0.122345, 0.124567};

#define TEST_VECTOR_SIZE (sizeof(test_tof_down) / sizeof(test_tof_down[0]))

void run_golden_vector_test(void)
{
    printf("=== Golden Vector Test ===\n");
    printf("Test size: %d\n", TEST_VECTOR_SIZE);
    
    measurement_init();
    
    for (int i = 0; i < TEST_VECTOR_SIZE; i++) {
        measurement_data_t data = {
            .timestamp_ms = i * 250,
            .tof_down_ticks = test_tof_down[i],
            .tof_up_ticks = test_tof_up[i],
            .temperature_c = test_temp[i],
            .valid_down = 1,
            .valid_up = 1,
            .status = 3
        };
        
        flow_result_t result;
        measurement_calculate(&data, &result);
        
        float error = result.velocity_m_s - expected_velocity[i];
        float error_percent = (error / expected_velocity[i]) * 100.0f;
        
        printf("GV,%d,%.6f,%.6f,%.6f,%.3f%%\n",
               i,
               expected_velocity[i],
               result.velocity_m_s,
               error,
               error_percent);
    }
    
    printf("=== Test Complete ===\n");
}

#endif /* GOLDEN_VECTOR_TEST */
```

## ⚠️ 注意事项

1. **根据实际 API 调整**
   - 上述代码中的 `sonar_measure()`, `get_system_time_ms()`, `read_temperature()` 等函数需要根据实际的硬件抽象层 API 调整

2. **浮点运算性能**
   - 如果龙芯 LS1D 没有 FPU，浮点运算可能很慢
   - 考虑使用定点运算优化性能
   - MATLAB 可以帮助确定定点精度要求

3. **编译选项**
   - 确保启用优化：`-O2` 或 `-O3`
   - 如果有 FPU：`-mhard-float`
   - 如果没有 FPU：`-msoft-float`

4. **内存占用**
   - 注意滤波缓冲区大小
   - 如果内存紧张，考虑减小 `FILTER_WINDOW_SIZE`

5. **时钟精度**
   - 确保系统时钟足够精确（至少 1ms 分辨率）
   - 累计体积误差与时钟精度直接相关
